#define _GNU_SOURCE
#include <stdio.h>
#include <stdint.h>
#include <string.h>
#include <dlfcn.h>

static const unsigned char key[4] = {0xde, 0xad, 0xbe, 0xef};

static int looks_like_flag(const unsigned char *s) {
    int braces = 0;
    for (int i = 0; i < 55; i++) {
        if (s[i] < 32 || s[i] > 126) return 0;
        if (s[i] == '{' || s[i] == '}') braces++;
    }
    return braces >= 2;
}

static void try_decode(const unsigned char *enc, const char *tag) {
    unsigned char out[56];
    for (int i = 0; i < 55; i++)
        out[i] = enc[i] ^ key[i % 4];
    out[55] = 0;
    if (looks_like_flag(out))
        printf("[%s] %s\n", tag, out);
}

void run(void) {
    void *(*g)(char*) = dlsym(RTLD_DEFAULT, "guess");
    if (g) {
        char *p = g("x");
        if (p) {
            char buf[56];
            memcpy(buf, p + 64, 55);
            buf[55] = 0;
            if (looks_like_flag((unsigned char*)buf))
                printf("[guess] %s\n", buf);
            else
                printf("[guess raw] %.55s\n", buf);
        }
    }

    FILE *f = fopen("/proc/self/maps", "r");
    if (!f) return;

    char line[512];
    while (fgets(line, sizeof line, f)) {
        unsigned long start, end;
        char perms[8];
        if (sscanf(line, "%lx-%lx %7s", &start, &end, perms) != 3)
            continue;
        if (perms[0] != 'r') continue;

        unsigned char *p = (unsigned char*)start;
        unsigned char *q = (unsigned char*)end;

        for (unsigned char *cur = p; cur + 4 <= q; cur++) {
            if (cur[0] == 0xde && cur[1] == 0xad &&
                cur[2] == 0xbe && cur[3] == 0xef) {
                for (int off = -80; off <= 80; off++) {
                    if (off == 0) continue;
                    unsigned char *enc = cur + off;
                    if (enc >= p && enc + 55 <= q)
                        try_decode(enc, "scan");
                }
            }
        }
    }
    fclose(f);
}

vaslang
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_FUNCS 65536
#define MAX_NAME 256

typedef struct {
    char name[MAX_NAME];
    char *tokens[10000];
    int ntokens;
} Func;

Func *funcs;
int nfuncs = 0;

int find_func(const char *name) {
    for (int i = 0; i < nfuncs; i++) {
        if (strcmp(funcs[i].name, name) == 0)
            return i;
    }
    return -1;
}

void parse_file(const char *filename) {
    funcs = calloc(MAX_FUNCS, sizeof(Func));
    if (!funcs) { perror("calloc"); exit(1); }

    FILE *f = fopen(filename, "r");
    if (!f) { perror("fopen"); exit(1); }
    char *line = malloc(1 << 20);
    while (fgets(line, 1 << 20, f)) {
        char *p = line;
        while (*p && isspace((unsigned char)*p)) p++;
        char *end = p + strlen(p);
        while (end > p && isspace((unsigned char)end[-1])) *--end = 0;
        if (*p == 0 || strcmp(p, "main!") == 0) continue;

        char *colon = strchr(p, ':');
        if (!colon) continue;
        *colon = 0;
        char *name = p;
        char *body = colon + 1;

        int idx = find_func(name);
        if (idx < 0) {
            idx = nfuncs++;
            strncpy(funcs[idx].name, name, MAX_NAME - 1);
            funcs[idx].ntokens = 0;
        }
        char *save = NULL;
        char *tok = strtok_r(body, " \t", &save);
        while (tok && funcs[idx].ntokens < 10000) {
            funcs[idx].tokens[funcs[idx].ntokens++] = strdup(tok);
            tok = strtok_r(NULL, " \t", &save);
        }
    }
    free(line);
    fclose(f);
}

int main(int argc, char **argv) {
    const char *filename = argc > 1 ? argv[1] : "inp.txt";
    parse_file(filename);

    int main_idx = find_func("main");
    if (main_idx < 0) { printf("main not found\n"); return 1; }

    unsigned char mem[1024] = {0};
    int OFFSET = 128;
    int vsp = OFFSET;

    struct Frame { int func_idx; int pc; } stack[1024];
    int sp = 0;
    stack[sp++] = (struct Frame){main_idx, 0};

    while (sp > 0) {
        struct Frame *fr = &stack[sp - 1];
        Func *fn = &funcs[fr->func_idx];
        if (fr->pc >= fn->ntokens) { sp--; continue; }

        char *tok = fn->tokens[fr->pc++];

        if (strcmp(tok, "+") == 0) {
            vsp--; int l = mem[vsp];
            vsp--; int r = mem[vsp];
            mem[vsp] = (l + r) % 251; vsp++;
        } else if (strcmp(tok, "*") == 0) {
            vsp--; int l = mem[vsp];
            vsp--; int r = mem[vsp];
            mem[vsp] = (l * r) % 251; vsp++;
        } else if (strcmp(tok, ":") == 0) {
            vsp--; int l = mem[vsp];
            mem[vsp] = l; vsp++;
            mem[vsp] = l; vsp++;
        } else if (strcmp(tok, ".") == 0) {
            vsp--;
        } else {
            char *endp;
            long val = strtol(tok, &endp, 10);
            if (*endp == 0 && endp != tok) {
                mem[vsp++] = (unsigned char)(val & 0xFF);
            } else {
                int idx = find_func(tok);
                if (idx >= 0 && sp < 1024) {
                    stack[sp++] = (struct Frame){idx, 0};
                }
            }
        }
    }

    printf("Raw bytes: ");
    for (int i = OFFSET; i < vsp; i++) printf("%d ", mem[i]);
    printf("\n");

    printf("String: ");
    for (int i = OFFSET; i < vsp; i++) {
        if (mem[i] == 0) break;
        putchar(mem[i]);
    }
    putchar('\n');
    return 0;
}
