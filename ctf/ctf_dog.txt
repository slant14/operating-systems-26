4. 02ScatteredDog:
cd Videos
cd 'Scatteder dog'
nano sol.c

#include <stdio.h>
int main() {
    FILE *out = fopen("scattered_dog.png", "wb");
    if (!out) return 1;
    char buffer[8192];
    size_t bytes;
    for (int i = -50; i <= 49; i++) {
        char filename[10];
        sprintf(filename, "%d", i);

        FILE *in = fopen(filename, "rb");
        if (!in) continue;

        while ((bytes = fread(buffer, 1, sizeof(buffer), in)) > 0) {
            fwrite(buffer, 1, bytes, out);
        }
        fclose(in);
    }
    fclose(out);
    return 0;
}

gcc sol.c -o sol
./sol
scp "e.gilfanov-dog@osctf.nnk.moe:/home/ctf/go_to_opt/Videos/Scattered dog/scattered_dog.png" ~/OS/operating-systems-26-week1/
