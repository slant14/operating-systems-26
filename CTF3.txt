    while (*s) {

    int count = 0;

    int ppid;

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <pthread.h>
#include <sys/wait.h>

#define BINARY "./decrapinator-x86_64-linux"
#define COPIES_PER_SHARD 6
#define MAX_SHARDS 1024

char *results[MAX_SHARDS];
int solved_count = 0;
int total_shards = 0;
pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;

typedef struct {
    int shard_idx;
} TaskArg;

void* worker(void *arg) {
    int idx = ((TaskArg*)arg)->shard_idx;
    free(arg);

    char cmd;
    snprintf(cmd, sizeof(cmd), "%s decrypt shards/%d.shard 2>/dev/null", BINARY, idx);

    while (1) {
        pthread_mutex_lock(&lock);
        if (results[idx] != NULL) {
            pthread_mutex_unlock(&lock);
            return NULL;
        }
        pthread_mutex_unlock(&lock);

        FILE *fp = popen(cmd, "r");
        if (!fp) {
            usleep(10000);
            continue;
        }

        char buf = {0};
        size_t len = fread(buf, 1, sizeof(buf) - 1, fp);
        int status = pclose(fp);

        while (len > 0 && (buf[len - 1] == '\n' || buf[len - 1] == '\r' || buf[len - 1] == ' ')) {
            buf[--len] = '\0';
        }

        if (WIFEXITED(status) && WEXITSTATUS(status) == 0 && len > 0) {
            pthread_mutex_lock(&lock);
            if (results[idx] == NULL) {
                results[idx] = strdup(buf);
                solved_count++;
                if (solved_count == total_shards) {
                    printf("\n=== ГОТОВЫЙ ФЛАГ ===\n");
                    FILE *clip = popen("clip.exe 2>/dev/null", "w");
                    for (int i = 0; i < total_shards; i++) {
                        printf("%s", results[i]);
                        if (clip) fputs(results[i], clip);
                    }
                    printf("\n====================\n");
                    if (clip && pclose(clip) == 0) {
                        printf("[+] Флаг автоматически скопирован в буфер обмена (Ctrl+V)!\n");
                    }
                    fflush(stdout);
                    _exit(0);
                }
            }
            pthread_mutex_unlock(&lock);
            return NULL;
        }
    }
    return NULL;
}

int main() {
    char path;
    while (total_shards < MAX_SHARDS) {
        snprintf(path, sizeof(path), "shards/%d.shard", total_shards);
        if (access(path, F_OK) == 0) {
            total_shards++;
        } else {
            break;
        }
    }

    if (total_shards == 0) {
        fprintf(stderr, "Шарды в папке shards/ не найдены!\n");
        return 1;
    }

    printf("Найдено шардов: %d. Запускаю %d потоков...\n", total_shards, total_shards * COPIES_PER_SHARD);

    int total_threads = total_shards * COPIES_PER_SHARD;
    pthread_t *threads = malloc(sizeof(pthread_t) * total_threads);

    int t = 0;
    for (int i = 0; i < total_shards; i++) {
        for (int c = 0; c < COPIES_PER_SHARD; c++) {
            TaskArg *arg = malloc(sizeof(TaskArg));
            arg->shard_idx = i;
            pthread_create(&threads[t++], NULL, worker, arg);
        }
    }

    for (int i = 0; i < total_threads; i++) {
        pthread_join(threads[i], NULL);
    }

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <time.h>
#include <string.h>

int main() {
    int fd = open("secrets", O_RDONLY);
    if (fd < 0) return 1;

    char *buf = malloc(65536);
    char *tail = calloc(4096, 1);
    size_t tail_len = 0;
    time_t start = time(NULL);

    while (1) {
        if (time(NULL) - start >= 3) {
            pid_t pid = fork();
            if (pid > 0) _exit(0);
            if (pid < 0) return 1;
            start = time(NULL);
        }

        ssize_t n = read(fd, buf, 65536);
        if (n <= 0) break;

        if (n >= 4096) {
            memcpy(tail, buf + n - 4096, 4096);
            tail_len = 4096;
        } else {
            if (tail_len + n > 4096) {
                size_t shift = tail_len + n - 4096;
                memmove(tail, tail + shift, tail_len - shift);
                tail_len -= shift;
            }
            memcpy(tail + tail_len, buf, n);
            tail_len += n;
        }
    }

    FILE *out = fopen("/tmp/flag.txt", "wb");
    if (out) {
        fwrite(tail, 1, tail_len, out);
        fclose(out);
    }
    return 0;
}
